cmake_minimum_required(VERSION 3.16)
project(tsfp_offline_shader_host LANGUAGES C CXX)
if(NOT TSFP_OPENSHACCCG_SOURCE)
    message(FATAL_ERROR "Use tools/build_offline_shaders.py with verified source")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OPENSHACCG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(OPENSHACCG_ENABLE_GLSLANG ON CACHE BOOL "" FORCE)
set(OPENSHACCG_ENABLE_SPIRV_TOOLS ON CACHE BOOL "" FORCE)
set(OPENSHACCG_ENABLE_SPIRV_CROSS ON CACHE BOOL "" FORCE)
# Resolve shader-front-end dependencies only from verified extracted packages.
# The host compiler/standard library/Threads remain platform dependencies.
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
set(glslang_DIR "${TSFP_SHADER_PREFIX}/lib/x86_64-linux-gnu/cmake/glslang" CACHE PATH "" FORCE)
set(SPIRV-Tools_DIR "${TSFP_SHADER_PREFIX}/lib/x86_64-linux-gnu/cmake/SPIRV-Tools" CACHE PATH "" FORCE)
set(SPIRV-Tools-opt_DIR "${TSFP_SHADER_PREFIX}/lib/x86_64-linux-gnu/cmake/SPIRV-Tools-opt" CACHE PATH "" FORCE)
find_package(SPIRV-Tools CONFIG REQUIRED PATHS "${SPIRV-Tools_DIR}" NO_DEFAULT_PATH)
find_package(SPIRV-Tools-opt CONFIG REQUIRED PATHS "${SPIRV-Tools-opt_DIR}" NO_DEFAULT_PATH)
find_package(glslang CONFIG REQUIRED PATHS "${glslang_DIR}" NO_DEFAULT_PATH)
set(OPENSHACCG_SPIRV_CROSS_INCLUDE_DIR "${TSFP_SHADER_PREFIX}/include" CACHE PATH "" FORCE)
set(OPENSHACCG_SPIRV_CROSS_CORE_LIBRARY "${TSFP_SHADER_PREFIX}/lib/x86_64-linux-gnu/libspirv-cross-core.a" CACHE FILEPATH "" FORCE)
function(tsfp_require_dependency_path path)
    get_filename_component(owned "${TSFP_SHADER_PREFIX}" REALPATH)
    get_filename_component(actual "${path}" REALPATH)
    file(RELATIVE_PATH relative "${owned}" "${actual}")
    if(NOT EXISTS "${path}" OR relative MATCHES "^\\.\\." OR IS_ABSOLUTE "${relative}")
        message(FATAL_ERROR "Shader dependency resolved outside the locked prefix")
    endif()
endfunction()
foreach(dependency glslang::glslang glslang::SPIRV glslang::glslang-default-resource-limits SPIRV-Tools-opt SPIRV-Tools-static)
    if(NOT TARGET ${dependency})
        message(FATAL_ERROR "Missing locked shader dependency target ${dependency}")
    endif()
    get_target_property(configurations ${dependency} IMPORTED_CONFIGURATIONS)
    foreach(configuration IN LISTS configurations)
        string(TOUPPER "${configuration}" configuration)
        get_target_property(location ${dependency} IMPORTED_LOCATION_${configuration})
        if(location)
            tsfp_require_dependency_path("${location}")
        endif()
    endforeach()
endforeach()
tsfp_require_dependency_path("${OPENSHACCG_SPIRV_CROSS_INCLUDE_DIR}/spirv_cross/spirv_cross.hpp")
tsfp_require_dependency_path("${OPENSHACCG_SPIRV_CROSS_CORE_LIBRARY}")
add_subdirectory("${TSFP_OPENSHACCCG_SOURCE}" compiler)
add_executable(tsfp_shader_compile main.cpp)
target_link_libraries(tsfp_shader_compile PRIVATE openshacccg_core)
target_compile_options(tsfp_shader_compile PRIVATE -Wall -Wextra -Werror -pedantic)

target_include_directories(openshacccg_core PRIVATE "${TSFP_SHADER_PREFIX}/include")
target_link_directories(tsfp_shader_compile BEFORE PRIVATE "${TSFP_SHADER_PREFIX}/lib/x86_64-linux-gnu")
